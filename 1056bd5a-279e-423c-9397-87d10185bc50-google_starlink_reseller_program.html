Google Apps Script Live Webhook (Google Drive + Gmail)

Copy this pre-configured Google Apps Script code into your Google Apps Script editor bound to your sailabs.wc@gmail.com account. When deployed as a Web App (Access: Anyone), signups will automatically create a document in Google Drive and send an email notification to sailabs.wc@gmail.com. 

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // 1. Target Email Notification for SA-iLabs
    var targetEmail = "sailabs.wc@gmail.com";
    var subject = "NEW GOOGLE STARLINK REGISTRATION: " + data.refId + " (" + data.organization + ")";
    
    var body = "NEW STARLINK ENTERPRISE SIGNUP RECEIVED\n\n" +
               "Reference ID: " + data.refId + "\n" +
               "Applicant Name: " + data.fullName + "\n" +
               "Organization: " + data.organization + "\n" +
               "Email: " + data.email + "\n" +
               "Phone: " + data.phone + "\n" +
               "Location: " + data.location + "\n\n" +
               "PACKAGE DETAILS:\n" +
               "Package: New Client Starlink Onboarding\n" +
               "Installation: " + data.installationType + " (R" + data.installationPrice + ")\n" +
               "Monthly Subscription: R950\n" +
               "Annual Subscription: 12 x R950 = R11,400\n" +
               "Payment Terms: Payment upon receipt\n" +
               "Accepted: Immediate Release Payments, PayShap, or Cash\n" +
               "Amount Due Upon Receipt: " + data.totalPrice + "\n\n" +
               "BANKING DETAILS:\n" +
               "Capitec Bank | Sailabs | Entrepreneur / Current Acc\n" +
               "Branch: 470010 | Account #: 2490686905\n\n" +
               "LEAD RESELLER:\n" +
               "Christo Botha (Google LLC Acknowledged Reseller, Cape Town, ZA)\n\n" +
               "Timestamp: " + new Date().toString();

    GmailApp.sendEmail(targetEmail, subject, body);

    // 2. Save directly into Google Drive Folder
    var folderName = "SA-iLabs Starlink Registrations";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    var fileName = "Registration_" + data.refId + "_" + data.organization.replace(/[^a-zA-Z0-9]/g, "_") + ".txt";
    folder.createFile(fileName, body);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", refId: data.refId }))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
